cmake_minimum_required(VERSION 3.22.1)
project(maple_jni LANGUAGES CXX)
add_library(maple_core STATIC IMPORTED)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(RUST_TARGET aarch64-linux-android)
elseif(ANDROID_ABI STREQUAL "x86_64")
    set(RUST_TARGET x86_64-linux-android)
else()
    message(FATAL_ERROR "Unsupported Maple Android ABI: ${ANDROID_ABI}")
endif()
set_target_properties(maple_core PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/../../../../native/target/${RUST_TARGET}/release/libmaple_android.a")
add_library(maple_jni SHARED bridge.cpp)
target_compile_features(maple_jni PRIVATE cxx_std_17)
target_link_libraries(maple_jni PRIVATE maple_core android nativewindow jnigraphics log dl m)
target_link_options(maple_jni PRIVATE "-Wl,-z,max-page-size=16384")
