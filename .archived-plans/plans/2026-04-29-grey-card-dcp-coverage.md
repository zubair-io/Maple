**Goal:** Cover the DCP code paths (`color/dcp.rs`, `hsm.rs`, `profile_tone_curve.rs`, `profile_gain_table_map.rs`) with closed-form synthetic-input tests — Phase 1 extends `SyntheticGreyDng` with optional dual-CM / FM / PTC fields against real Hasselblad data; Phase 2 adds a new `SyntheticColorChart` to exercise HSM, ProfileLookTable, and ProfileGainTableMap.

Predictors mirror production stage math; drift tests pin them at 1e-6 against the production code.

```rust
//! One-shot Hasselblad DCP extractor. Run when test_0000.DNG changes
//! (rare). Dumps ColorMatrix1, ColorMatrix2, AsShotNeutral, and the
//! calibration illuminant codes in source-pasteable form.
//!
//!   cargo run --release -p raw-core --example extract-dcp -- \
//!       test-fixtures/raws/test_0000.DNG

use raw_core::decode::decode_bytes;
use std::env;

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let path = env::args().nth(1).ok_or("usage: extract-dcp <path-to-dng>")?;
    let bytes = std::fs::read(&path)?;
    let raw = decode_bytes(&bytes, "dng")?;

    println!("// Auto-generated by examples/extract-dcp.rs from {}", path);
    println!("// camera: {} {}", raw.camera_make, raw.camera_model);
    println!();
    let mut entries: Vec<_> = raw.color_matrices.iter().collect();
    entries.sort_by_key(|(illum, _)| format!("{:?}", illum));
    for (illum, mat) in entries {
        println!("// ColorMatrix for {:?}:", illum);
        println!("pub const COLOR_MATRIX_{:?}: [[f32; 3]; 3] = [", illum);
        for row in 0..3 {
            print!("    [");
            for col in 0..3 {
                if col > 0 { print!(", "); }
                print!("{:.6}", mat.data[row][col]);
            }
            println!("],");
        }
        println!("];");
        println!();
    }
    println!("pub const AS_SHOT_NEUTRAL: [f32; 3] = [{:.6}, {:.6}, {:.6}];",
        raw.as_shot_neutral[0], raw.as_shot_neutral[1], raw.as_shot_neutral[2]);
    Ok(())
}
```

```rust
/// Convert a 3×3 f32 matrix to (i32, i32) srational pairs.
/// Numerator scaled by 1_000_000, denominator = 1_000_000 — gives ~6
/// decimal digits of precision, matches what real DCP writers do.
fn matrix_to_srationals(m: [[f32; 3]; 3]) -> Vec<(i32, i32)> {
    const SCALE: f32 = 1_000_000.0;
    let mut out = Vec::with_capacity(9);
    for row in 0..3 {
        for col in 0..3 {
            let n = (m[row][col] * SCALE).round() as i32;
            out.push((n, SCALE as i32));
        }
    }
    out
}

/// Convert a [f32; 3] to (u32, u32) rational triples.
fn vec3_to_rationals(v: [f32; 3]) -> Vec<(u32, u32)> {
    const SCALE: f32 = 1_000_000.0;
    v.iter()
        .map(|&x| ((x * SCALE).round() as u32, SCALE as u32))
        .collect()
}
```

```rust
pub fn predict_tone_curve(scene: f32, control_points: &[(f32, f32)]) -> f32 {
    if control_points.len() < 2 { return scene; }
    let x = scene.clamp(control_points[0].0, control_points[control_points.len() - 1].0);

    // Find the bracket [i, i+1] such that points[i].0 <= x <= points[i+1].0.
    let mut i = 0;
    while i + 1 < control_points.len() && control_points[i + 1].0 < x {
        i += 1;
    }
    if i + 1 >= control_points.len() {
        return control_points.last().unwrap().1;
    }

    let (x0, y0) = control_points[i];
    let (x1, y1) = control_points[i + 1];
    let dx = x1 - x0;
    if dx.abs() < 1e-9 { return y0; }
    let t = (x - x0) / dx;

    // Monotonic-cubic Hermite (Fritsch-Carlson). Compute tangents:
    let tangent = |k: usize| -> f32 {
        if k == 0 {
            (control_points[1].1 - control_points[0].1) / (control_points[1].0 - control_points[0].0)
        } else if k + 1 >= control_points.len() {
            (control_points[k].1 - control_points[k - 1].1) / (control_points[k].0 - control_points[k - 1].0)
        } else {
            let h_prev = control_points[k].0 - control_points[k - 1].0;
            let h_next = control_points[k + 1].0 - control_points[k].0;
            let s_prev = (control_points[k].1 - control_points[k - 1].1) / h_prev;
            let s_next = (control_points[k + 1].1 - control_points[k].1) / h_next;
            if s_prev * s_next <= 0.0 { 0.0 }
            else {
                3.0 * (h_prev + h_next) /
                    ((2.0 * h_next + h_prev) / s_prev + (h_next + 2.0 * h_prev) / s_next)
            }
        }
    };
    let m0 = tangent(i);
    let m1 = tangent(i + 1);
    let h00 = (1.0 + 2.0 * t) * (1.0 - t) * (1.0 - t);
    let h10 = t * (1.0 - t) * (1.0 - t);
    let h01 = t * t * (3.0 - 2.0 * t);
    let h11 = t * t * (t - 1.0);
    h00 * y0 + h10 * dx * m0 + h01 * y1 + h11 * dx * m1
}
```

```rust
    /// Walk the chart row-major; for each pixel, look up which patch it
    /// belongs to, get that patch's target scene-linear color, and write
    /// the 16-bit raw value at the CFA position.
    fn build_strip(&self) -> Vec<u8> {
        let w = self.width() as usize;
        let h = self.height() as usize;
        let mut buf = Vec::with_capacity(w * h * 2);
        for y in 0..h {
            for x in 0..w {
                let v = self.raw_at(x as u32, y as u32);
                write_u16_le(&mut buf, v);
            }
        }
        buf
    }

    fn raw_at(&self, x: u32, y: u32) -> u16 {
        let (col, row) = self.patch_at(x, y);
        let target = self.patches[row as usize][col as usize];
        // Synthesize per CFA position. WB is identity ([1,1,1] AsShotNeutral),
        // so post-WB scene-linear == raw / white_level. raw_R = R * WL etc.
        let target_per_channel = match self.cfa.color_at(x, y) {
            0 => target[0],
            1 => target[1],
            2 => target[2],
            _ => unreachable!(),
        };
        (target_per_channel.clamp(0.0, 1.0) * 65535.0).round() as u16
    }

    /// Map pixel (x, y) to (col, row). Pixels in guard bands return
    /// the nearest patch (no separate "guard" target — guard regions just
    /// repeat the last patch on each side).
    fn patch_at(&self, x: u32, y: u32) -> (u32, u32) {
        let stride = self.patch_size + self.guard;
        let col = (x / stride).min(self.cols() - 1);
        let row = (y / stride).min(self.rows() - 1);
        (col, row)
    }

    fn cfa_pattern_bytes(&self) -> Vec<u8> {
        match self.cfa {
            CfaPattern::Rggb => vec![0, 1, 1, 2],
            CfaPattern::Bggr => vec![2, 1, 1, 0],
            CfaPattern::Grbg => vec![1, 0, 2, 1],
            CfaPattern::Gbrg => vec![1, 2, 0, 1],
            CfaPattern::LinearRgb => panic!("SyntheticColorChart with LinearRgb is unsupported"),
        }
    }

    /// Read patch `(col, row)` from a developed `Image`. Returns the per-
    /// channel mean over the patch's interior, skipping the outer 4
    /// pixels on every side to absorb demosaic bleed.
    pub fn read_patch_mean(&self, image: &Image, col: usize, row: usize) -> [f32; 3] {
        let stride = (self.patch_size + self.guard) as usize;
        let x0 = col * stride + 4;
        let y0 = row * stride + 4;
        let inner = self.patch_size as usize - 8;
        let w = image.width as usize;
        let mut sums = [0.0_f64; 3];
        let mut n: u64 = 0;
        for dy in 0..inner {
            for dx in 0..inner {
                let i = (y0 + dy) * w + (x0 + dx);
                if i >= image.pixels.len() { continue; }
                let p = image.pixels[i];
                sums[0] += p[0] as f64;
                sums[1] += p[1] as f64;
                sums[2] += p[2] as f64;
                n += 1;
            }
        }
        let nn = n.max(1) as f64;
        [(sums[0] / nn) as f32, (sums[1] / nn) as f32, (sums[2] / nn) as f32]
    }
}
```

```rust
pub fn predict_radial_gain(input: f32, radius_norm: f32, gain_values: &[f32]) -> f32 {
    if gain_values.is_empty() { return input; }
    if gain_values.len() == 1 { return input * gain_values[0]; }

    let r = radius_norm.clamp(0.0, 1.0);
    let n = gain_values.len();
    let scaled = r * (n as f32 - 1.0);
    let i = scaled.floor() as usize;
    if i + 1 >= n {
        return input * gain_values[n - 1];
    }
    let t = scaled - i as f32;
    let g = gain_values[i] + t * (gain_values[i + 1] - gain_values[i]);
    input * g
}
```

```rust
pub fn predict_hsm_cell(
    input_rgb: [f32; 3],
    dims: &[u32; 3],
    data: &[f32],
) -> [f32; 3] {
    // 1. RGB → HSV
    let r = input_rgb[0]; let g = input_rgb[1]; let b = input_rgb[2];
    let max = r.max(g).max(b);
    let min = r.min(g).min(b);
    let v = max;
    let s = if max > 1e-9 { (max - min) / max } else { 0.0 };
    let h = if (max - min).abs() < 1e-9 {
        0.0
    } else if (max - r).abs() < 1e-9 {
        ((g - b) / (max - min)).rem_euclid(6.0) * 60.0
    } else if (max - g).abs() < 1e-9 {
        ((b - r) / (max - min) + 2.0) * 60.0
    } else {
        ((r - g) / (max - min) + 4.0) * 60.0
    };

    // 2. Lattice index for (h, s, v)
    let (hd, sd, vd) = (dims[0] as i32, dims[1] as i32, dims[2] as i32);
    let h_idx = ((h / 360.0) * hd as f32) as i32;
    let s_idx = (s * (sd - 1) as f32).round() as i32;
    let v_idx = if vd == 1 { 0 } else { (v * (vd - 1) as f32).round() as i32 };

    // (Production uses trilinear interp; nearest-neighbour is a starting
    // point — switch to trilinear if drift tests fail.)
    let h_idx = h_idx.clamp(0, hd - 1) as usize;
    let s_idx = s_idx.clamp(0, sd - 1) as usize;
    let v_idx = v_idx.clamp(0, vd - 1) as usize;

    let entry_idx = ((h_idx * sd as usize + s_idx) * vd as usize + v_idx) * 3;
    let hue_delta = data[entry_idx];
    let sat_scale = data[entry_idx + 1];
    let val_scale = data[entry_idx + 2];

    // 3. Apply deltas
    let h_new = (h + hue_delta).rem_euclid(360.0);
    let s_new = (s * sat_scale).clamp(0.0, 1.0);
    let v_new = (v * val_scale).clamp(0.0, f32::MAX);

    // 4. HSV → RGB
    let c = v_new * s_new;
    let h_p = h_new / 60.0;
    let x = c * (1.0 - (h_p.rem_euclid(2.0) - 1.0).abs());
    let (rp, gp, bp) = match h_p as i32 {
        0 => (c, x, 0.0),
        1 => (x, c, 0.0),
        2 => (0.0, c, x),
        3 => (0.0, x, c),
        4 => (x, 0.0, c),
        _ => (c, 0.0, x),
    };
    let m = v_new - c;
    [rp + m, gp + m, bp + m]
}
```

- [ ] **Step 1: Build a 1D radial GainTableMap blob**

The DNG ProfileGainTableMap header is 44 bytes per spec; data is `V × H × planes × 4` bytes. For a 1D radial gain we use V=1, H=5, planes=1, origin (0.5, 0.5), spacing (0, 0.5). Append to `chart_dcp_phase2.rs`:

```rust
fn make_test_radial_gain_bytes() -> Vec<u8> {
    // ProfileGainTableMap header layout per DNG 1.6 spec:
    //   [u32 version=1]
    //   [u32 flags=0]
    //   [u32 mapPointsV=1]
    //   [u32 mapPointsH=5]
    //   [f32 mapSpacingV=0.0]
    //   [f32 mapSpacingH=0.25]
    //   [f32 mapOriginV=0.5]
    //   [f32 mapOriginH=0.0]
    //   [u32 mapPlanes=1]
    //   [f32 gain[5] = {1.5, 1.3, 1.0, 0.85, 0.7}]
    let mut buf = Vec::with_capacity(44 + 5 * 4);
    buf.extend_from_slice(&1u32.to_le_bytes());      // version
    buf.extend_from_slice(&0u32.to_le_bytes());      // flags
    buf.extend_from_slice(&1u32.to_le_bytes());      // V
    buf.extend_from_slice(&5u32.to_le_bytes());      // H
    buf.extend_from_slice(&0.0f32.to_le_bytes());    // spacingV
    buf.extend_from_slice(&0.25f32.to_le_bytes());   // spacingH
    buf.extend_from_slice(&0.5f32.to_le_bytes());    // originV
    buf.extend_from_slice(&0.0f32.to_le_bytes());    // originH
    buf.extend_from_slice(&1u32.to_le_bytes());      // planes
    for g in [1.5_f32, 1.3, 1.0, 0.85, 0.7] {
        buf.extend_from_slice(&g.to_le_bytes());
    }
    buf
}

#[test]
fn gain_table_radial() {
    let mut chart = SyntheticColorChart::default();
    chart.gain_table_map_bytes = Some(make_test_radial_gain_bytes());

    let img = render_chart(&chart);

    // White patch (row 3 col 0) is in upper-left → high radius.
    // Neutral 5 (row 3 col 3) is closer to center → lower radius.
    // Their gain ratio should match the predicted ratio.
    let upper_left = chart.read_patch_mean(&img, 0, 3);
    let center_ish = chart.read_patch_mean(&img, 3, 3);

    // Estimate radius (normalised) for each patch using the chart center.
    let cx = chart.width() as f32 / 2.0;
    let cy = chart.height() as f32 / 2.0;
    let stride = (chart.patch_size + chart.guard) as f32;
    let patch_xy = |col: u32, row: u32| -> (f32, f32) {
        (col as f32 * stride + chart.patch_size as f32 / 2.0,
         row as f32 * stride + chart.patch_size as f32 / 2.0)
    };
    let r_norm = |x: f32, y: f32| -> f32 {
        ((x - cx).powi(2) + (y - cy).powi(2)).sqrt()
            / (cx.max(cy))
    };
    let (ulx, uly) = patch_xy(0, 3);
    let (cix, ciy) = patch_xy(3, 3);
    let r_ul = r_norm(ulx, uly);
    let r_ci = r_norm(cix, ciy);

    let lut = [1.5_f32, 1.3, 1.0, 0.85, 0.7];
    let target = 0.949_f32;
    let predicted_ul = predict_radial_gain(target, r_ul, &lut);
    let predicted_ci = predict_radial_gain(target, r_ci, &lut);
    let predicted_ratio = predicted_ul / predicted_ci;
    let observed_ratio = upper_left[0] / center_ish[0];

    assert!((observed_ratio - predicted_ratio).abs() <= 0.05,
        "radial gain ratio mismatch: observed = {}, predicted = {}",
        observed_ratio, predicted_ratio);
}
```
